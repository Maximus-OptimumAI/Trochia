       ```ts
       // keys are derived from a const tuple so adding a key later requires updating
       // BOTH the tuple AND the type. Construction goes through `buildTraceMetadata`
       // which rejects extra keys at compile time. Test Case 8 also asserts the

         'source_type',
         'source_id',
         'chunk_count',
         'input_type',
         'token_count',
         'embedding_model_version',

         // already prevent this, but a runtime strip protects against any future
         // refactor that loosens the type)
         const out: TraceMetadata = {};
         for (const key of TRACE_METADATA_KEYS) {
           if (meta[key] !== undefined) out[key] = meta[key];
         }
         return out;

         embeddings: number[][]; // shape: [texts.length][1024]
         tokenCount: number;

         latencyMs: number;

         dim: VOYAGE_DIM,

         async embed(input: VoyageEmbedInput): Promise<VoyageEmbedResult> {
           if (input.texts.length === 0) {

               method: 'POST',
               headers: {
                 'Content-Type': 'application/json',
                 Authorization: `Bearer ${env.VOYAGE_API_KEY}`,

                 input: input.texts,
                 input_type: input.inputType,
                 truncation: false,

               signal: controller.signal,

               'VOYAGE_BATCH_FAILED',
               `voyage ${response.status}: ${body}`,

             }

           // Both `input` and `output` are constructed through buildTraceMetadata
           // which strips any key not in TRACE_METADATA_KEYS. Test Case 8 asserts

               token_count: json.usage.total_tokens,
               embedding_model_version: VOYAGE_MODEL,
               latency_ms: latencyMs,
             }),
           });

             embeddings: embeddingsOut,
             tokenCount: json.usage.total_tokens,
             modelVersion: VOYAGE_MODEL,
             latencyMs,
       ```

       ```ts
       /**
        * Deterministic text chunker (Plan 02-04 / KNW-04b).
        *
        * Pure function. Zero I/O. Zero logger calls. Same input → same output, byte-for-byte.
        *
        * Strategy: token-budgeted sliding window with token-aware overlap.
        *   - target chunk size: 800 tokens (config: opts.size)
        *   - overlap: 200 tokens (config: opts.overlap)
        *   - boundary: prefer the LAST sentence-end (.!?) inside the window; fall back to whitespace; fall back to mid-token cut as last resort
        *
        * Token counting (cycle-1 review LOW #1 — drift caveat):
        *   - if `@anthropic-ai/tokenizer` is installed → use `countTokens(text)` for exact count
        *   - else fallback heuristic: 1 token ≈ 4 characters (an APPROXIMATION; the
        *     `tokenCount` field on every Chunk is therefore an ESTIMATE under the heuristic
        *     path. Documented in `tasks/lessons.md` and surfaced on the type as
        *     `tokenCount /* estimated under heuristic; exact under @anthropic-ai/tokenizer */`.
        *     Voyage's 32K-token input limit is far above any 800-token-target chunk under
        *     either path, so drift here does not threaten correctness — only the cost-tracking
        *     fidelity in Plan 02-07's OBS-COST-01 budgeting). Replaced with real tokenizer
        *     in Plan 02-05 if eval shows drift.
        *
        * Output rows:
        *   { text: string; idx: number; tokenCount: number }
        *
        * idx is 0-indexed and contiguous (0, 1, 2, ..., N-1). Empty input throws.
        */

       export type ChunkOptions = {
         size: number;       // target tokens per chunk
         overlap: number;    // tokens of overlap between adjacent chunks
       };

       export type Chunk = {
         text: string;
         idx: number;
         tokenCount: number;
       };

       export const DEFAULT_CHUNK_OPTIONS: ChunkOptions = { size: 800, overlap: 200 };

       export function chunkText(text: string, opts: ChunkOptions = DEFAULT_CHUNK_OPTIONS): Chunk[] {
         if (!text || text.trim().length === 0) {
           throw new Error('CHUNK_INPUT_EMPTY');
         }
         if (opts.size <= 0 || opts.overlap < 0 || opts.overlap >= opts.size) {
           throw new Error('CHUNK_OPTS_INVALID');
         }

         const tokensPerChar = 1 / 4; // heuristic fallback
         const sizeChars = Math.floor(opts.size / tokensPerChar);    // 3200 chars for 800 tokens
         const overlapChars = Math.floor(opts.overlap / tokensPerChar); // 800 chars

         const chunks: Chunk[] = [];
         let cursor = 0;
         let idx = 0;

         while (cursor < text.length) {
           const windowEnd = Math.min(cursor + sizeChars, text.length);
           let cutAt = windowEnd;

           // Prefer sentence boundary inside the last 200 chars of the window
           if (cutAt < text.length) {
             const tail = text.slice(cutAt - 200, cutAt);
             const lastSentence = Math.max(tail.lastIndexOf('. '), tail.lastIndexOf('! '), tail.lastIndexOf('? '));
             if (lastSentence > 0) {
               cutAt = cutAt - 200 + lastSentence + 1; // include the punctuation
             } else {
               const lastSpace = tail.lastIndexOf(' ');
               if (lastSpace > 0) cutAt = cutAt - 200 + lastSpace;
             }
           }

           const chunkContent = text.slice(cursor, cutAt).trim();
           if (chunkContent.length === 0) break;

           chunks.push({
             text: chunkContent,
             idx,
             tokenCount: Math.ceil(chunkContent.length * tokensPerChar),
           });
           idx += 1;

           if (cutAt >= text.length) break;
           cursor = Math.max(cutAt - overlapChars, cursor + 1);
         }

         return chunks;
       }
       ```

       ```ts
        * embed-memory Inngest function (Plan 02-04 / KNW-04b).
        *
        * Trigger: 'memory.confirmed' event emitted by src/server/routers/memory.ts
        * immediately after a successful confirmDraft mutation.
        *

        * is a Voyage call cost). Final-failure path writes a Sentry breadcrumb and

        *
        * TOCTOU lock semantics (cycle-1 review fix HIGH #5 — explicit boundary):
        *   Inngest `step.run` boundaries do NOT participate in one shared DB transaction.
        *   Each `step.run` is independent; only the body of a given step.run can run inside
        *   a DB transaction. The atomic guarantee in this function is therefore narrow:
        *
        *     Step 1 (load-and-chunk)              — NO DB lock; ordinary SELECT
        *     Step 2 (voyage-embed-batch-*)         — NO DB lock; HTTP only
        *     Step 3 (upsert-embeddings)           — INSIDE a single DB transaction:
        *        3a. SELECT business_memory.lastUpdatedAt FOR SHARE — row-level read lock
        *        3b. Compare against event.data.lastUpdatedAt (captured at event-emit time)
        *        3c. If drifted → EMBED_TOCTOU_STALE breadcrumb + return 0 (newer
        *            memory.confirmed will re-trigger from scratch)
        *        3d. If match → DELETE existing embeddings rows for (account, memory,
        *            source_id, model_version) and INSERT freshly-computed chunks.
        *
        *   The atomicity invariant is "compare current lastUpdatedAt immediately before
        *   write," NOT "lock the row across the whole function." Between Steps 1 and 3,
        *   a concurrent confirmDraft CAN update business_memory.lastUpdatedAt; that's
        *   exactly the race the TOCTOU compare catches. Step 3's transaction holds the
        *   row read-locked from 3a through 3d, so the DELETE-then-INSERT happens against
        *   a stable lastUpdatedAt observation.
        *

        *   Inside the upsert-embeddings transaction, AFTER the lastUpdatedAt match,
        *   the function executes:

        *        AND source_id=? AND embedding_model_version=?
        *     2. INSERT ... (the freshly computed chunks)

        *   under the same dedup key would otherwise leave stale chunk_text in place
        *   (ON CONFLICT DO NOTHING does NOT update). It also removes orphaned higher
        *   `chunk_idx` rows when new content has fewer chunks than the prior version.

           concurrency: { limit: 3, key: 'event.data.accountId' },
           retries: 2,

         async ({ event, step }) => {

                   message: 'EMBED_ROW_GONE',
                   level: 'warning',
                   data: { accountId, businessMemoryId },

                 return 0;

                   message: 'EMBED_TOCTOU_STALE',
                   level: 'info',
                   data: {
                     accountId,
                     businessMemoryId,
                     emittedAt: new Date(emittedAt).toISOString(),
                     currentAt: current.lastUpdatedAt.toISOString(),
                   },
                 });
                 return 0; // newer memory.confirmed will re-trigger

               // dedup key. We delete the prior chunks for (account, source_type='memory',

               // transaction with the FOR SHARE lock from the lastUpdatedAt re-read,

                     tokenCount: e.tokenCount,
                   })),
                 );

               return embedded.length;
             });
           });

           return { ok: true, chunks: written };
         },
       );
       ```

       ```ts
       // src/inngest/functions/corpus-sync.ts (top of file)
       import { createHash } from 'node:crypto';

       // RFC 4122 UUID v5 namespace for Trochia corpus documents. This is a static,
       // arbitrary UUID — we are not part of any well-known namespace tree. Generated
       // once via `node -e "console.log(require('crypto').randomUUID())"` and pinned.
       const CORPUS_NAMESPACE = '6f6e0e6f-7adc-4b3f-a3a9-3a3d1d4b9d33';

       /** Deterministic UUID v5 from (CORPUS_NAMESPACE, slug). No `uuid` package required. */
       function corpusSourceUuid(slug: string): string {
         // Pack the namespace UUID to its 16 raw bytes
         const nsBytes = Buffer.from(CORPUS_NAMESPACE.replace(/-/g, ''), 'hex');
         // SHA-1 over (namespace_bytes || utf8(slug))
         const hash = createHash('sha1').update(nsBytes).update(slug, 'utf8').digest();
         // First 16 bytes; set version (v5 = 0101) and variant (RFC 4122 = 10xx) bits
         const bytes = Buffer.from(hash.subarray(0, 16));
         bytes[6] = (bytes[6] & 0x0f) | 0x50; // version 5
         bytes[8] = (bytes[8] & 0x3f) | 0x80; // RFC 4122 variant
         const hex = bytes.toString('hex');
         return `${hex.slice(0, 8)}-${hex.slice(8, 12)}-${hex.slice(12, 16)}-${hex.slice(16, 20)}-${hex.slice(20, 32)}`;
       }
       ```

       ```ts
       import { qaGrounding } from './checks/qa-grounding';
       import { cacheHit } from './checks/cache-hit';
       import type { EvalCheck, EvalSuiteResult } from './types';
       import { writeFileSync, appendFileSync } from 'node:fs';

       const CHECKS: EvalCheck[] = [extractionFloor, qaGrounding, cacheHit];

       export async function runEvalSuite(): Promise<EvalSuiteResult> {
         const results = await Promise.all(CHECKS.map((c) => c.run()));
         const anyFail = results.some((r) => r.status === 'fail');
         const exitCode: 0 | 1 = anyFail ? 1 : 0;

         const summary = [
           '## Trochia eval harness',
           '',
           '| Check | Status | Reason |',
           '|---|---|---|',
           ...results.map((r) => `| \`${r.id}\` | ${badge(r.status)} | ${r.reason} |`),
           '',
           `**Exit code:** ${exitCode}`,
         ].join('\n');

         // CI-friendly JSON to stdout
         process.stdout.write(JSON.stringify({ checks: results, exitCode }, null, 2) + '\n');

         // Markdown summary to GITHUB_STEP_SUMMARY when running in CI
         if (process.env.GITHUB_STEP_SUMMARY) {
           appendFileSync(process.env.GITHUB_STEP_SUMMARY, summary + '\n');
         }

         // Local JSON artifact
         writeFileSync('eval-report.json', JSON.stringify({ checks: results, exitCode }, null, 2));

         return { checks: results, exitCode, summary };
       }

       function badge(s: 'pending' | 'pass' | 'fail'): string {
         return s === 'pass' ? '✅ pass' : s === 'fail' ? '❌ fail' : '⏳ pending';
       }

       // CLI entrypoint
       if (require.main === module) {
         runEvalSuite().then((r) => process.exit(r.exitCode));
       }
       ```
